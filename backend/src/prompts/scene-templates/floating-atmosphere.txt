// UTILITY — Always include this file. Customize colors to match brand palette.
import { motion, useTransform, useSpring } from "framer-motion";
import { useEffect, useRef, useState, useCallback } from "react";

function generateParticles(count) {
  return Array.from({ length: count }, (_, i) => ({
    id: i,
    x: Math.random() * 100,
    y: Math.random() * 100,
    size: 1 + Math.random() * 3,
    speed: 15 + Math.random() * 25,
    opacity: 0.05 + Math.random() * 0.15,
    delay: Math.random() * 10,
  }));
}

export default function FloatingAtmosphere() {
  const [mousePos, setMousePos] = useState({ x: 0.5, y: 0.5 });
  const particles = useRef(generateParticles(40)).current;

  const handleMouseMove = useCallback((e) => {
    setMousePos({ x: e.clientX / window.innerWidth, y: e.clientY / window.innerHeight });
  }, []);

  useEffect(() => {
    window.addEventListener("mousemove", handleMouseMove, { passive: true });
    return () => window.removeEventListener("mousemove", handleMouseMove);
  }, [handleMouseMove]);

  const smoothX = useSpring(mousePos.x, { stiffness: 30, damping: 30 });
  const smoothY = useSpring(mousePos.y, { stiffness: 30, damping: 30 });

  useEffect(() => { smoothX.set(mousePos.x); smoothY.set(mousePos.y); }, [mousePos]);

  const glowX = useTransform(smoothX, [0, 1], ["20%", "80%"]);
  const glowY = useTransform(smoothY, [0, 1], ["20%", "80%"]);

  /* CONTENT: Replace hsl values with brand primary and accent colors */
  return (
    <div className="fixed inset-0 pointer-events-none z-[1]" style={{ background: "var(--brand-bg, hsl(252, 10%, 3%))" }}>
      <motion.div className="absolute w-[600px] h-[600px] rounded-full"
        style={{ left: glowX, top: glowY, x: "-50%", y: "-50%",
          background: "radial-gradient(circle, var(--brand-glow-a, hsl(203, 72%, 38%, 0.04)) 0%, transparent 70%)" }} />
      <motion.div className="absolute w-[900px] h-[900px] rounded-full"
        style={{ top: "20%", left: "30%",
          background: "radial-gradient(circle, var(--brand-glow-a, hsl(203, 72%, 38%, 0.03)) 0%, transparent 60%)" }}
        animate={{ x: [0, 80, -40, 0], y: [0, -60, 40, 0], scale: [1, 1.2, 0.9, 1] }}
        transition={{ duration: 30, repeat: Infinity, ease: "easeInOut" }} />
      <motion.div className="absolute w-[700px] h-[700px] rounded-full"
        style={{ bottom: "10%", right: "20%",
          background: "radial-gradient(circle, var(--brand-glow-b, hsl(197, 71%, 38%, 0.025)) 0%, transparent 60%)" }}
        animate={{ x: [0, -60, 50, 0], y: [0, 50, -30, 0], scale: [1, 0.85, 1.15, 1] }}
        transition={{ duration: 25, repeat: Infinity, ease: "easeInOut" }} />
      {particles.map((p) => (
        <motion.div key={p.id} className="absolute rounded-full"
          style={{ width: p.size, height: p.size, left: `${p.x}%`,
            background: `var(--brand-particle, hsl(203, 72%, 55%, ${p.opacity}))` }}
          animate={{ top: [`${p.y}%`, `${p.y - 30}%`, `${p.y}%`], opacity: [p.opacity * 0.5, p.opacity, p.opacity * 0.5], x: [0, (Math.random() - 0.5) * 40, 0] }}
          transition={{ duration: p.speed, repeat: Infinity, ease: "easeInOut", delay: p.delay }} />
      ))}
      <div className="absolute inset-0 flex items-center justify-center overflow-hidden">
        {[0, 1, 2].map((i) => (
          <motion.div key={i} className="absolute rounded-full border"
            style={{ width: 400 + i * 200, height: 400 + i * 200,
              borderColor: `var(--brand-ring, hsl(203, 72%, 38%, ${0.015 - i * 0.003}))` }}
            animate={{ rotate: [0, i % 2 === 0 ? 360 : -360] }}
            transition={{ duration: 80 + i * 20, repeat: Infinity, ease: "linear" }} />
        ))}
      </div>
    </div>
  );
}
