import { useInView } from "./useInView";

const ease = [0.22, 1, 0.36, 1];

export default function IconDeconstruction() {
  const { ref, isInView } = useInView(0.3);

  /* CONTENT: Replace with client icon elements and their meanings */
  const elements = [
    { symbol: "e", color: "var(--brand-primary)", desc: "The foundation — a single letterform that carries the entire identity." },
    { symbol: "3", color: "var(--brand-accent)", desc: "Hidden in the negative space — a nod to the brand's origins." },
    { symbol: "∞", color: "text-foreground/40", desc: "The infinite loop — continuous evolution, never static." },
  ];

  return (
    <section ref={ref} className="flow-section relative px-8">
      <div className="max-w-5xl w-full relative z-10 grid grid-cols-1 md:grid-cols-2 gap-16 items-center">
        <motion.div className="flex flex-col items-center perspective-container"
          initial={{ opacity: 0, scale: 0.9 }}
          animate={isInView ? { opacity: 1, scale: 1 } : {}}
          transition={{ duration: 1.2, delay: 0.3, ease }}>
          {/* CONTENT: Replace with brand icon — image or large symbol */}
